def create_db(name):
    db = sqlite3.connect(ROOT / (name + '.db'))
    db.row_factory = sqlite3.Row
    db.execute('PRAGMA journal_mode=WAL')
    db.execute('PRAGMA foreign_keys=ON')
    _ensure_schema(db)
    dispatcher.ensure_dispatch_table(db)
    return db

def create_run(db, name, records=None, only_report=False, lead_id=None):
    staging = ROOT / name
    staging.mkdir(exist_ok=True)
    (staging / 'report.md').write_text('# Review fixture\n')
    (staging / 'run.json').write_text('{}')
    if not only_report:
        for filename in dispatcher.REQUIRED_ARTIFACTS:
            if filename.endswith('.jsonl'):
                (staging / filename).write_text('')
        if records:
            (staging / 'candidate_findings.jsonl').write_text(''.join(json.dumps(r)+'\n' for r in records))
    cur = db.execute('INSERT INTO dispatch_runs (run_type, job_type, target, status, review_required, expected_artifacts, staging_dir, task_contract_json, lead_id) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)', ('trace_entity','trace_entity','Example Corp','completed',1,json.dumps(dispatcher.REQUIRED_ARTIFACTS),str(staging),json.dumps({'profile_id':'review-test'}),lead_id))
    run_id = cur.lastrowid
    run = db.execute('SELECT * FROM dispatch_runs WHERE id=?',(run_id,)).fetchone()
    inspection = dispatcher.inspect_staging_bundle(db, run, {}, update_db=True)
    db.execute("UPDATE dispatch_staging SET review_status='approved' WHERE run_id=?", (run_id,))
    db.commit()
    return run_id, inspection
