person_a TEXT NOT NULL, person_b TEXT NOT NULL,

description TEXT, strength TEXT DEFAULT 'medium' CHECK(strength IN ('strong','medium','weak','circumstantial')),
date_range TEXT, finding_id INTEGER REFERENCES findings(id), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
agent_run_id TEXT
